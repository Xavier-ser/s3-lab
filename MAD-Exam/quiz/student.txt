package com.example.quizapp;

import android.content.Intent;
import android.content.SharedPreferences;
import android.os.Bundle;
import android.widget.Button;
import android.widget.LinearLayout;
import android.widget.RadioButton;
import android.widget.RadioGroup;
import android.widget.TextView;
import android.widget.Toast;

import androidx.appcompat.app.AppCompatActivity;

public class QuizActivity extends AppCompatActivity {

    LinearLayout quizLayout;
    Button submitButton;
    Button logoutButton;

    RadioGroup[] groups;

    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);

        setContentView(R.layout.activity_quiz);

        quizLayout = findViewById(R.id.quizLayout);
        submitButton = findViewById(R.id.submitButton);
        logoutButton = findViewById(R.id.logoutButton);

        int numberOfQuestions =
                CreatorActivity.questions.size();

        if (numberOfQuestions == 0) {

            TextView noQuestion = new TextView(this);

            noQuestion.setText(
                    "No quiz questions available.\nCreator must add questions first."
            );

            noQuestion.setTextSize(18);

            quizLayout.addView(noQuestion);

            submitButton.setEnabled(false);

            return;
        }

        groups = new RadioGroup[numberOfQuestions];

        for (int i = 0; i < numberOfQuestions; i++) {

            TextView questionText =
                    new TextView(this);

            questionText.setText(
                    (i + 1) + ". " +
                    CreatorActivity.questions.get(i)
            );

            questionText.setTextSize(18);

            questionText.setPadding(
                    0, 20, 0, 10
            );

            quizLayout.addView(questionText);

            RadioGroup group =
                    new RadioGroup(this);

            String[] opts =
                    CreatorActivity.options.get(i);

            for (int j = 0; j < 4; j++) {

                RadioButton radioButton =
                        new RadioButton(this);

                radioButton.setText(opts[j]);
                radioButton.setTextSize(16);

                group.addView(radioButton);
            }

            quizLayout.addView(group);

            groups[i] = group;
        }

        submitButton.setOnClickListener(
                v -> calculateScore()
        );

        logoutButton.setOnClickListener(
                v -> logout()
        );
    }

    private void calculateScore() {

        int score = 0;

        for (int i = 0; i < groups.length; i++) {

            int selectedId =
                    groups[i].getCheckedRadioButtonId();

            if (selectedId == -1) {
                continue;
            }

            RadioButton selected =
                    findViewById(selectedId);

            int selectedIndex =
                    groups[i].indexOfChild(selected);

            int correctIndex =
                    CreatorActivity.answers.get(i);

            if (selectedIndex == correctIndex) {
                score++;
            }
        }

        int total = groups.length;

        // Save marks
        SharedPreferences preferences =
                getSharedPreferences(
                        "QuizData",
                        MODE_PRIVATE
                );

        SharedPreferences.Editor editor =
                preferences.edit();

        editor.putInt("score", score);
        editor.putInt("total", total);

        editor.apply();

        Toast.makeText(
                this,
                "Your Score: " + score + "/" + total,
                Toast.LENGTH_LONG
        ).show();

        submitButton.setEnabled(false);
    }

    private void logout() {

        Intent intent =
                new Intent(
                        QuizActivity.this,
                        MainActivity.class
                );

        intent.setFlags(
                Intent.FLAG_ACTIVITY_NEW_TASK |
                Intent.FLAG_ACTIVITY_CLEAR_TASK
        );

        startActivity(intent);

        finish();
    }
}
