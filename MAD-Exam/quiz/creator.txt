package com.example.quizapp;

import android.content.Intent;
import android.os.Bundle;
import android.widget.Button;
import android.widget.EditText;
import android.widget.RadioGroup;
import android.widget.Toast;

import androidx.appcompat.app.AppCompatActivity;

import java.util.ArrayList;

public class CreatorActivity extends AppCompatActivity {

    EditText question, option1, option2, option3, option4;

    RadioGroup correctGroup;

    Button addButton;
    Button logoutButton;

    static ArrayList<String> questions = new ArrayList<>();
    static ArrayList<String[]> options = new ArrayList<>();
    static ArrayList<Integer> answers = new ArrayList<>();

    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);

        setContentView(R.layout.activity_creator);

        question = findViewById(R.id.question);
        option1 = findViewById(R.id.option1);
        option2 = findViewById(R.id.option2);
        option3 = findViewById(R.id.option3);
        option4 = findViewById(R.id.option4);

        correctGroup = findViewById(R.id.correctGroup);

        addButton = findViewById(R.id.addButton);
        logoutButton = findViewById(R.id.logoutButton);

        // ADD QUESTION
        addButton.setOnClickListener(v -> addQuestion());

        // LOGOUT
        logoutButton.setOnClickListener(v -> logout());
    }

    private void addQuestion() {

        String q = question.getText().toString().trim();
        String o1 = option1.getText().toString().trim();
        String o2 = option2.getText().toString().trim();
        String o3 = option3.getText().toString().trim();
        String o4 = option4.getText().toString().trim();

        if (q.isEmpty() ||
                o1.isEmpty() ||
                o2.isEmpty() ||
                o3.isEmpty() ||
                o4.isEmpty()) {

            Toast.makeText(
                    this,
                    "Fill all fields",
                    Toast.LENGTH_SHORT
            ).show();

            return;
        }

        int selectedId =
                correctGroup.getCheckedRadioButtonId();

        if (selectedId == -1) {

            Toast.makeText(
                    this,
                    "Select correct answer",
                    Toast.LENGTH_SHORT
            ).show();

            return;
        }

        int correctAnswer;

        if (selectedId == R.id.correct1) {
            correctAnswer = 0;
        } else if (selectedId == R.id.correct2) {
            correctAnswer = 1;
        } else if (selectedId == R.id.correct3) {
            correctAnswer = 2;
        } else {
            correctAnswer = 3;
        }

        questions.add(q);

        String[] optionArray = {
                o1, o2, o3, o4
        };

        options.add(optionArray);

        answers.add(correctAnswer);

        Toast.makeText(
                this,
                "Question Added Successfully",
                Toast.LENGTH_SHORT
        ).show();

        // Clear fields
        question.setText("");
        option1.setText("");
        option2.setText("");
        option3.setText("");
        option4.setText("");

        correctGroup.clearCheck();
    }

    private void logout() {

        Intent intent = new Intent(
                CreatorActivity.this,
                MainActivity.class
        );

        intent.setFlags(
                Intent.FLAG_ACTIVITY_NEW_TASK |
                Intent.FLAG_ACTIVITY_CLEAR_TASK
        );

        startActivity(intent);

        finish();
    }
}
