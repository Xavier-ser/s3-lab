package com.example.facultyleave;

import android.app.DatePickerDialog;
import android.content.SharedPreferences;
import android.os.Bundle;
import android.widget.ArrayAdapter;
import android.widget.Button;
import android.widget.EditText;
import android.widget.Spinner;
import android.widget.TextView;
import android.widget.Toast;

import androidx.appcompat.app.AppCompatActivity;

import java.util.Calendar;

public class ApplyLeaveActivity extends AppCompatActivity {

    TextView selectedDate;
    Spinner leaveType;
    EditText reason;
    Button applyButton;

    SharedPreferences preferences;

    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);

        setContentView(R.layout.activity_apply_leave);

        selectedDate = findViewById(R.id.selectedDate);
        leaveType = findViewById(R.id.leaveType);
        reason = findViewById(R.id.reason);
        applyButton = findViewById(R.id.applyButton);

        preferences = getSharedPreferences(
                "LeaveData",
                MODE_PRIVATE
        );

        // Spinner values
        String[] types = {
                "Casual",
                "Medical",
                "Duty"
        };

        ArrayAdapter<String> adapter =
                new ArrayAdapter<>(
                        this,
                        android.R.layout.simple_spinner_dropdown_item,
                        types
                );

        leaveType.setAdapter(adapter);

        // DatePicker
        selectedDate.setOnClickListener(v -> {

            Calendar calendar = Calendar.getInstance();

            int year = calendar.get(
                    Calendar.YEAR
            );

            int month = calendar.get(
                    Calendar.MONTH
            );

            int day = calendar.get(
                    Calendar.DAY_OF_MONTH
            );

            DatePickerDialog datePicker =
                    new DatePickerDialog(
                            ApplyLeaveActivity.this,
                            (view, selectedYear,
                             selectedMonth,
                             selectedDay) -> {

                                String date =
                                        selectedDay + "/"
                                                + (selectedMonth + 1)
                                                + "/"
                                                + selectedYear;

                                selectedDate.setText(date);
                            },
                            year,
                            month,
                            day
                    );

            datePicker.show();
        });

        // Apply Leave
        applyButton.setOnClickListener(v -> {

            String date =
                    selectedDate.getText().toString();

            String type =
                    leaveType.getSelectedItem().toString();

            String leaveReason =
                    reason.getText().toString().trim();

            if (date.equals("Select Date")) {

                Toast.makeText(
                        this,
                        "Please select a date",
                        Toast.LENGTH_SHORT
                ).show();

                return;
            }

            if (leaveReason.isEmpty()) {

                Toast.makeText(
                        this,
                        "Please enter reason",
                        Toast.LENGTH_SHORT
                ).show();

                return;
            }

            SharedPreferences.Editor editor =
                    preferences.edit();

            editor.putString(
                    "leaveDate",
                    date
            );

            editor.putString(
                    "leaveType",
                    type
            );

            editor.putString(
                    "reason",
                    leaveReason
            );

            // Initial status
            editor.putString(
                    "status",
                    "Pending"
            );

            editor.apply();

            Toast.makeText(
                    this,
                    "Leave application submitted",
                    Toast.LENGTH_SHORT
            ).show();

            finish();
        });
    }
}
