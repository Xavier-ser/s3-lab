package com.example.alumninetwork;

import android.content.Intent;
import android.content.SharedPreferences;
import android.os.Bundle;
import android.view.Menu;
import android.view.MenuItem;
import android.widget.Button;
import android.widget.EditText;
import android.widget.Toast;

import androidx.appcompat.app.AppCompatActivity;

public class MainActivity extends AppCompatActivity {

    EditText etName, etYear, etCompany;
    Button btnContinue;

    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);

        setContentView(R.layout.activity_main);

        etName = findViewById(R.id.etName);
        etYear = findViewById(R.id.etYear);
        etCompany = findViewById(R.id.etCompany);
        btnContinue = findViewById(R.id.btnContinue);

        btnContinue.setOnClickListener(v -> {

            // Read input values
            String name = etName.getText().toString().trim();
            String year = etYear.getText().toString().trim();
            String company = etCompany.getText().toString().trim();

            // Validate name
            if (name.isEmpty()) {
                etName.setError("Enter full name");
                etName.requestFocus();
                return;
            }

            // Validate graduation year
            if (year.isEmpty()) {
                etYear.setError("Enter graduation year");
                etYear.requestFocus();
                return;
            }

            int graduationYear;

            try {
                graduationYear = Integer.parseInt(year);

                if (graduationYear < 1950 ||
                        graduationYear > 2035) {

                    etYear.setError(
                            "Enter a valid graduation year"
                    );

                    etYear.requestFocus();
                    return;
                }

            } catch (NumberFormatException e) {

                etYear.setError(
                        "Enter a valid graduation year"
                );

                etYear.requestFocus();
                return;
            }

            // Validate company
            if (company.isEmpty()) {
                etCompany.setError("Enter current company");
                etCompany.requestFocus();
                return;
            }

            // Save alumnus name using SharedPreferences
            SharedPreferences preferences =
                    getSharedPreferences(
                            "AlumniData",
                            MODE_PRIVATE
                    );

            SharedPreferences.Editor editor =
                    preferences.edit();

            editor.putString("alumnus_name", name);
            editor.apply();

            // Bundle remaining information
            Bundle bundle = new Bundle();

            bundle.putInt(
                    "graduation_year",
                    graduationYear
            );

            bundle.putString(
                    "current_company",
                    company
            );

            // Explicit Intent
            Intent intent =
                    new Intent(
                            MainActivity.this,
                            DashboardActivity.class
                    );

            intent.putExtras(bundle);

            startActivity(intent);
        });
    }

    // Create Options Menu
    @Override
    public boolean onCreateOptionsMenu(Menu menu) {

        getMenuInflater().inflate(
                R.menu.main_menu,
                menu
        );

        return true;
    }

    // Handle Options Menu
    @Override
    public boolean onOptionsItemSelected(MenuItem item) {

        if (item.getItemId() == R.id.menu_clear) {

            etName.setText("");
            etYear.setText("");
            etCompany.setText("");

            Toast.makeText(
                    this,
                    "Form cleared",
                    Toast.LENGTH_SHORT
            ).show();

            return true;
        }

        return super.onOptionsItemSelected(item);
    }
}
